cmake_minimum_required(VERSION 3.16)
project(brehm_cpp CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BREHM_SANITIZE "build with AddressSanitizer (and UBSan where available)" OFF)

# GMP with its C++ interface (gmpxx), e.g. from vcpkg, Homebrew, or libgmp-dev.
find_path(GMP_INCLUDE_DIR gmpxx.h REQUIRED)
find_library(GMP_LIBRARY gmp REQUIRED)
find_library(GMPXX_LIBRARY gmpxx REQUIRED)

add_executable(brehm_cpp main.cpp brehm.cpp check.cpp orders.cpp)
target_include_directories(brehm_cpp SYSTEM PRIVATE ${GMP_INCLUDE_DIR})
target_link_libraries(brehm_cpp PRIVATE ${GMPXX_LIBRARY} ${GMP_LIBRARY})
install(TARGETS brehm_cpp RUNTIME DESTINATION bin)
if(WIN32)  # vcpkg keeps the GMP DLLs in bin/ next to lib/; install them beside the executable
  get_filename_component(GMP_LIB_DIR ${GMP_LIBRARY} DIRECTORY)
  file(GLOB GMP_DLLS ${GMP_LIB_DIR}/../bin/gmp*.dll)
  install(FILES ${GMP_DLLS} DESTINATION bin)
endif()
if(MSVC)
  target_compile_options(brehm_cpp PRIVATE /W4 /EHsc /external:W0)
  if(BREHM_SANITIZE)
    target_compile_options(brehm_cpp PRIVATE /fsanitize=address)
  endif()
else()
  target_compile_options(brehm_cpp PRIVATE -Wall -Wextra)
  if(BREHM_SANITIZE)
    target_compile_options(brehm_cpp PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(brehm_cpp PRIVATE -fsanitize=address,undefined)
  endif()
endif()

enable_testing()
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME examples COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/run_tests.py $<TARGET_FILE:brehm_cpp>)
  add_test(NAME fuzz COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/fuzz.py 200 1 $<TARGET_FILE:brehm_cpp>)
  add_test(NAME malformed COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/malformed.py $<TARGET_FILE:brehm_cpp>)
  add_test(NAME figure COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/figure_tests.py $<TARGET_FILE:brehm_cpp> --random 20)
  # the stored certificates, checked by the standalone stdlib tools in isolated mode (no project imports)
  foreach(name regression_line regression_fold)
    add_test(NAME ${name}_ref_check COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/tools/ref_check.py
             ${CMAKE_CURRENT_SOURCE_DIR}/examples/${name}_cert.json --instance ${CMAKE_CURRENT_SOURCE_DIR}/examples/${name}.json)
    add_test(NAME ${name}_brehm_eval COMMAND ${Python3_EXECUTABLE} -I ${CMAKE_CURRENT_SOURCE_DIR}/tools/brehm_eval.py
             ${CMAKE_CURRENT_SOURCE_DIR}/examples/${name}_cert.json --instance ${CMAKE_CURRENT_SOURCE_DIR}/examples/${name}.json)
    add_test(NAME ${name}_check COMMAND brehm_cpp check ${CMAKE_CURRENT_SOURCE_DIR}/examples/${name}_cert.json)
  endforeach()
endif()
